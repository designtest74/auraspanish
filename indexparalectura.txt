            activeMixSelection.push({ name, hex, dotId });
        }

        // Update visual slots
        const slot1 = document.getElementById('mix-slot-1');
        const slot2 = document.getElementById('mix-slot-2');
        
        slot1.style.backgroundColor = activeMixSelection[0] ? activeMixSelection[0].hex : 'transparent';
        slot1.textContent = activeMixSelection[0] ? '' : '?';
        
        slot2.style.backgroundColor = activeMixSelection[1] ? activeMixSelection[1].hex : 'transparent';
        slot2.textContent = activeMixSelection[1] ? '' : '?';

        document.getElementById('mix-btn').disabled = (activeMixSelection.length !== 2);
    }

    /* ==========================================================================
       5. MATCHING PAIRS LOGIC
       ========================================================================== */
    let firstFlippedCard = null;
    let isBoardLocked = false;

    function renderMatching() {
        updateProgressBar(80);
        document.getElementById('match-clue').textContent = "🇬🇧 Match each Spanish word (e.g. Rojo) with its colour circle (🔴).";
        document.getElementById('match-feedback').classList.remove('show');
        document.getElementById('match-next').style.display = 'none';

        const board = document.getElementById('memory-board');
        board.innerHTML = '';
        state.matching.matchedPairs = 0;
        firstFlippedCard = null;
        isBoardLocked = false;

        const shuffledCards = [...matchCardPairs].sort(() => Math.random() - 0.5);

        shuffledCards.forEach(item => {
            const card = document.createElement('div');
            card.className = 'memory-card';
            card.dataset.id = item.id;
            card.dataset.val = item.val;
            card.innerHTML = `<span>${item.val}</span>`;
            card.onclick = () => flipMemoryCard(card);
            board.appendChild(card);
        });
    }

    function flipMemoryCard(card) {
        if (isBoardLocked || card === firstFlippedCard || card.classList.contains('matched')) return;

        card.classList.add('flipped');

        if (!firstFlippedCard) {
            firstFlippedCard = card;
            if (card.dataset.val.length > 2) speakSpanish(card.dataset.val);
            return;
        }

        // Second card flipped
        const secondCard = card;
        if (secondCard.dataset.val.length > 2) speakSpanish(secondCard.dataset.val);

        if (firstFlippedCard.dataset.id === secondCard.dataset.id) {
            firstFlippedCard.classList.add('matched');
            secondCard.classList.add('matched');
            firstFlippedCard = null;
            state.matching.matchedPairs++;
            state.score += 10;

            if (state.matching.matchedPairs === 6) {
                displayFeedback('match-feedback', true, `Superb! (¡Excelente!) You matched all 6 Spanish colour pairs!`);
                triggerConfetti();
                document.getElementById('match-next').style.display = 'inline-flex';
                updateProgressBar(100);
            }
        } else {
            isBoardLocked = true;
            displayFeedback('match-feedback', false, `Keep looking! Flip another pair.`);
            setTimeout(() => {
                firstFlippedCard.classList.remove('flipped');
                secondCard.classList.remove('flipped');
                firstFlippedCard = null;
                isBoardLocked = false;
                document.getElementById('match-feedback').classList.remove('show');
            }, 900);
        }
    }
